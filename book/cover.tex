% KDP paperback cover geometry (trim 6.69in x 9.61in, WHITE paper):
%   spine  = pages * 0.002252in
%   width  = 2*6.69 + 2*0.125 + spine
%   height = 9.61 + 2*0.125
% ONLY \PageCount below needs to change when the interior grows or shrinks;
% every dimension on this cover is derived from it.  \PageCount must equal
% the page count of kiss-book.pdf exactly, or KDP rejects the upload with
% "expected cover size is WxH but the submitted file size is ...".
\newcommand{\PageCount}{510}

% These use \fpeval (LaTeX kernel), NOT \pgfmathsetmacro: pgfmath's
% fixed-point arithmetic computes the spine as 1.0456in instead of
% 1.042676in, which is enough to fail KDP's cover-size check.
\newcommand{\TrimW}{6.69}
\newcommand{\TrimH}{9.61}
\newcommand{\Bleed}{0.125}
\edef\SpineW{\fpeval{\PageCount*0.002252}}
\edef\CoverW{\fpeval{2*\TrimW + 2*\Bleed + \SpineW}}
\edef\CoverH{\fpeval{\TrimH + 2*\Bleed}}
\edef\SpineL{\fpeval{\TrimW + \Bleed}}
\edef\SpineR{\fpeval{\SpineL + \SpineW}}
\edef\SpineC{\fpeval{\SpineL + \SpineW/2}}
\edef\FrontC{\fpeval{(\SpineR + \CoverW)/2}}
\edef\FrontRuleL{\fpeval{\FrontC - 1.5}}

\usepackage{geometry}
\geometry{paperwidth=\CoverW in, paperheight=\CoverH in, margin=0in}
